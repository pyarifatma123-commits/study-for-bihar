const updates = [
  "बिहार बोर्ड मैट्रिक परीक्षा 2025",
  "फ्री शौचालय योजना आवेदन फॉर्म",
  "RRB NTPC Recruitment 2025",
  "JEE Main 2026 Online Apply",
  "छात्रवृत्ति योजना 2025",
  "Admit Card Download Link",
  "University Result 2025",
  "Sarkari Yojna PDF"
];

function filterUpdates() {
  const input = document.getElementById("searchInput").value.toLowerCase();
  const resultBox = document.getElementById("updateResults");
  resultBox.innerHTML = "";

  const filtered = updates.filter(item => item.toLowerCase().includes(input));
  filtered.forEach(item => {
    const div = document.createElement("div");
    div.textContent = "📌 " + item;
    div.style.marginBottom = "10px";
    resultBox.appendChild(div);
  });

  if (filtered.length === 0 && input.length > 2) {
    resultBox.innerHTML = "❌ कोई जानकारी नहीं मिली";
  }
}
